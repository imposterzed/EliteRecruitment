# Assign the marshal traits to the base character (orchestrator)
assign_marshal_traits = {
  remove_marshal_negative_traits = yes

  assign_marshal_leadership_trait = yes
  assign_elite_chinese_marshal_trait = yes
  assign_marshal_warrior_lodge = yes

  assign_marshal_legendary_chance = yes

# Remove physical and martial drawbacks unbecoming of an elite marshal: a battle-leader
# must be hale and brave. giant and paranoid are deliberately kept (see inline notes).
remove_marshal_negative_traits = {
  remove_trait = weak
  remove_trait = slow
  remove_trait = imbecile
  remove_trait = dull
  remove_trait = inbred
  remove_trait = dwarf
  # giant not stripped -- net-positive for a marshal (combat rating)
  remove_trait = hunchback
  remove_trait = clubfooted
  remove_trait = craven
  # paranoid not stripped -- intrigue/honor trait, no martial impact

  # Elite addition: Slothful is -1 to all attributes
  remove_trait = slothful

  remove_lifestyle_trait_effect = yes
}

# Strip any leadership trait the base marshal was generated with, so our curated roll is the
# only standard leadership trait and there's room for a bonus (cap is 2). remove_trait on an
# absent trait is a no-op; build-renamed traits are cleared via the trait_compat utility.
remove_existing_leadership_traits = {
  # General
  remove_trait = aggressive_leader
  remove_trait = defensive_leader
  remove_trait = unyielding_leader
  remove_trait = flanker
  remove_trait = inspiring_leader
  remove_trait = trickster
  remove_trait = organizer
  remove_trait = siege_leader
  remove_trait = winter_soldier
  # holy_warrior is a holy-war trait but leader = yes, so it counts toward the cap
  remove_trait = holy_warrior

  # Unit-specific
  remove_trait = cavalry_leader
  remove_trait = heavy_infantry_leader
  remove_trait = light_foot_leader
  remove_trait = war_elephant_leader

  # Chinese (Jade Dragon)
  remove_trait = logistics_expert
  remove_trait = levy_coordinator
  remove_trait = sapper
  remove_trait = master_of_flame

  # Warrior lodge (Holy Fury)
  remove_trait = norse_leader
  remove_trait = tengri_leader
  remove_trait = baltic_leader
  remove_trait = finnish_leader
  remove_trait = slavic_leader
  remove_trait = west_african_leader
  remove_trait = zun_leader
  remove_trait = bon_leader
  remove_trait = hellenic_leader

  # Build-renamed leadership traits (Battlefield Terrain Master, Direct Leader, terrain)
  remove_elite_compat_leadership_traits = yes
}

# Guaranteed top martial education
set_elite_marshal_education = {
  clear_education_trait = yes
  add_trait = brilliant_strategist
}

    }

    # Terrain expertise
    10 = { assign_elite_desert_terrain_trait = yes } # Desert

    10 = { assign_elite_mountain_terrain_trait = yes } # Mountain
    10 = { assign_elite_rough_terrain_trait = yes } # Rough
  }
}

# Chinese (Jade Dragon) bonus: a Chinese-culture recruit has a ~50% chance at a second
# leadership trait (the cap is MAX_LEADERSHIP_TRAITS = 2), as vanilla grants these.
assign_elite_chinese_marshal_trait = {
  if = {
    limit = {
      has_dlc = "Jade Dragon"
      ROOT = { culture_group = chinese_group }
    }
    random_list = {
      100 = {}  # ~50% none
      25 = { trigger = { can_have_more_leadership_traits = yes } add_trait = logistics_expert } # Way of the Dog
      25 = { trigger = { can_have_more_leadership_traits = yes } add_trait = levy_coordinator } # Way of the Tiger
      25 = { trigger = { can_have_more_leadership_traits = yes } add_trait = sapper }           # Way of the Leopard
      25 = { trigger = { can_have_more_leadership_traits = yes } add_trait = master_of_flame }  # Way of the Dragon
    }
  }
}

# If the recruiting ruler belongs to a warrior lodge, sponsor the recruit into it,
# with a 25% chance they have already earned the lodge's command trait.
assign_marshal_warrior_lodge = {
  if = {
    limit = { ROOT = { in_warrior_lodge_trigger = yes } }
    ROOT = { society = { save_event_target_as = elite_ruler_lodge } }
    join_society = event_target:elite_ruler_lodge

    # ~25% effective chance at the lodge trait. A Chinese-culture ruler's recruit may have
    # already spent its 2nd leadership slot on a Chinese trait, so roll 50% for them
    # (50% Chinese-fail x 50% = ~25%) to match the rate other lodge members get.
    if = {
      limit = { ROOT = { culture_group = chinese_group } }
      random_list = {
        50 = {}
        50 = {
          trigger = { can_have_more_leadership_traits = yes }
          add_warrior_lodge_leader_trait_effect = yes
        }
      }
    }
    else = {
      random_list = {
        75 = {}
        25 = {
          trigger = { can_have_more_leadership_traits = yes }
          add_warrior_lodge_leader_trait_effect = yes
        }
      }
    }
  }
}

# Small chance the recruit is a holy-war veteran, with the trait matching their faith.
# Dharmic faiths have no such trait, so they are excluded.
#
# Inlined to bypass vanilla add_crusade_trait_effect: its per-branch `break = yes`
# propagates out of new_character and silently aborts ER.1. Five trait IDs differ
# between vanilla and CleanSlate, gated on cleanslate_active.
assign_marshal_crusade_trait = {
  if = {
    limit = { has_dharmic_religion_trigger = no }
    random = {
      chance = 15

      if = {
        limit = { religion_group = muslim }
        add_trait = mujahid
      }
      else_if = {
        limit = { religion_group = christian }
        add_trait = crusader
      }
      else_if = {
        limit = { religion_openly_norse_or_reformed_trigger = yes }
        add_trait = valhalla_bound
      }
      else_if = {
        limit = { religion_openly_tengri_or_reformed_trigger = yes }
        assign_elite_tengri_crusade_trait = yes
      }
      else_if = {
        limit = { religion_openly_finnish_or_reformed_trigger = yes }
        assign_elite_finnish_crusade_trait = yes
      }
      else_if = {
        limit = { religion_openly_slavic_or_reformed_trigger = yes }
        add_trait = peruns_chosen
      }
      else_if = {
        limit = { religion_openly_west_african_or_reformed_trigger = yes }
        add_trait = nyames_shield
      }
      else_if = {
        limit = { religion_openly_baltic_or_reformed_trigger = yes }
        assign_elite_baltic_crusade_trait = yes
      }
      else_if = {
        limit = { religion_openly_bon_or_reformed_trigger = yes }
        add_trait = kailash_guardian
      }
      else_if = {
        limit = { religion_openly_zun_or_reformed_trigger = yes }
        add_trait = sun_warrior
      }
      else_if = {
        limit = { religion_openly_aztec_or_reformed_trigger = yes }
        assign_elite_aztec_crusade_trait = yes
      }
      else_if = {
        limit = { religion_openly_hellenic_or_reformed_trigger = yes }
        add_trait = ares_own
      }
      else_if = {
        limit = { religion_group = zoroastrian_group }
        add_trait = gondi_shahansha
      }
      else_if = {
        limit = { religion_group = jewish_group }
        assign_elite_jewish_crusade_trait = yes
      }
    }
  }
}

# A veteran commander sometimes carries the marks of past battles. The penalty wounds
# are paired with a hardening trait so the roll is always worth getting.
assign_marshal_battle_wounds = {
  random = {
    chance = 15
    random_list = {
      # Battle scars -- no attribute penalty; the tier scales up with Holy Fury
      50 = {
        random_list = {
          50 = { add_trait = scarred }
          25 = { trigger = { has_dlc = "Holy Fury" } add_trait = scarred_mid }
          25 = { trigger = { has_dlc = "Holy Fury" } add_trait = scarred_high }
        }
      }
      # Lost an eye, but seasoned by it -- patient's +5 combat blunts the -10
      25 = {
        trigger = { has_dlc = "Reapers" }
        add_trait = one_eyed
        add_trait = patient
      }
      # Disfigured in the fighting, all the more wrathful for it
      25 = {
        trigger = { has_dlc = "Reapers" }
        add_trait = disfigured
        add_trait = wroth
      }
    }
  }
}

# Small chance of a legendary, named marshal if the ruler owns an inspire-commanders bloodline
assign_marshal_legendary_chance = {
  if = {
    limit = {  ROOT = {
      any_owned_bloodline = {
        has_bloodline_flag = bloodline_inspire_commanders
        bloodline_is_active_for = ROOT
      }
    } }
    random_list = {
      95 = {}
      5 = {
        assign_legendary_marshal = yes
        opinion = {
          modifier = opinion_legendary_commander
          years = 50
          who = ROOT
        }
      }
    }
  }
}
