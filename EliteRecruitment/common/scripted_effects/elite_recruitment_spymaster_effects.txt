# Assign the base spymaster attributes to the base character
assign_spymaster_base_attributes = {
  change_intrigue = 12
  change_diplomacy = 4
  change_learning = 4
  change_stewardship = 4
  change_martial = 4
}

# Assign the spymaster traits to the base character (orchestrator)
assign_spymaster_traits = {
  remove_spymaster_negative_traits = yes
  set_elite_spymaster_education = yes
  assign_spymaster_flavor = yes
  assign_religion_traits = yes

  set_character_flag = elite_recruit_spymaster
  save_event_target_as = invited_character
}

# Remove incompetence, frailty, and conspicuous traits unbecoming of an elite spymaster.
# A spy must blend in, so a memorable face, gait, or speech is a liability. paranoid is
# deliberately kept -- it is +2 intrigue, an asset for a spymaster.
remove_spymaster_negative_traits = {
  # Incompetence / frailty
  remove_trait = imbecile
  remove_trait = dull
  remove_trait = slow
  remove_trait = weak
  remove_trait = slothful

  # Conspicuousness -- a memorable figure makes a poor spy
  remove_trait = dwarf
  remove_trait = giant
  remove_trait = hunchback
  remove_trait = clubfooted
  remove_trait = harelip
  remove_trait = lisp
  remove_trait = stutter
  remove_trait = ugly
  remove_trait = inbred

  # Anti-spy personality
  remove_trait = craven
  remove_trait = honest    # -2 intrigue; honesty undercuts a spy
  remove_trait = trusting  # gullibility, the opposite of the paranoid we keep

  remove_lifestyle_trait_effect = yes
}

# Guaranteed top intrigue education
set_elite_spymaster_education = {
  clear_education_trait = yes
  add_trait = elusive_shadow
}
