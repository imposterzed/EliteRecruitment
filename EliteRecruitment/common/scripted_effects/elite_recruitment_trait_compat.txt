##################################################
##    Elite Recruitment Trait Compatibility     ##
##################################################
## Maps the flavor and leadership traits that   ##
## CleanSlate renames. The base uses vanilla    ##
## IDs; when the elite_recruitment_cleanslate   ##
## flag is set (by the CleanSlate patch) the    ##
## CleanSlate IDs are used instead. Per-trait   ##
## mapping is noted on each effect below.       ##
##################################################

# Physique flavor trait:
# Robust (vanilla) / Brawny (CleanSlate)
assign_elite_physique_trait = {
  if = {
    limit = { has_global_flag = elite_recruitment_cleanslate }
    add_trait = brawny
  }
  else = {
    add_trait = robust
  }
}

# Remove the build's renamed leadership traits (Battlefield Terrain Master, Direct Leader, terrain)
remove_elite_compat_leadership_traits = {
  if = {
    limit = { has_global_flag = elite_recruitment_cleanslate }
    # CleanSlate IDs
    remove_trait = battlefield_terrain_master
    remove_trait = direct_leader
    remove_trait = desert_expert
    remove_trait = flat_terrain_expert
    remove_trait = jungle_expert
    remove_trait = mountain_expert
    remove_trait = rough_terrain_expert
  }
  else = {
    # Vanilla IDs
    remove_trait = narrow_flank_leader
    remove_trait = experimenter
    remove_trait = desert_terrain_leader
    remove_trait = flat_terrain_leader
    remove_trait = jungle_terrain_leader
    remove_trait = mountain_terrain_leader
    remove_trait = rough_terrain_leader
  }
}
