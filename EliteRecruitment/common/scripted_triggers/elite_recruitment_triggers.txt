# --- Shared helpers ---

# Whether the elite_recruitment rule is anything but disabled (gates the
# Recruit an Elite Specialist decision and each option inside the menu)
elite_recruitment_enabled_trigger = {
  NOT = { has_game_rule = { name = elite_recruitment value = disabled } }
}

# Whether any male role's sex-gate passes. Used by the decision to auto-skip
# the gender picker when only one sex qualifies. Marshal/Steward/Chancellor/
# Spymaster all share "NOT matriarchal"; the male Chaplain widens this when a
# matriarchal faith still ordains men.
elite_recruitment_any_male_role_eligible_trigger = {
  OR = {
    NOT = { has_religion_feature = religion_matriarchal }
    religion_allows_male_temple_holders = yes
  }
}

# Whether any female role's sex-gate passes. Union of every female decision's
# gate: realm_allows_female_courtiers covers gender=all / matriarchal / equal
# / feature_bon; the rest are status_of_women_4 (the secular ladder threshold),
# the spymaster's wider pagan-non-patriarchal / Cathar / Messalian carve-outs,
# and the female chaplain's faith-ordination clause.
elite_recruitment_any_female_role_eligible_trigger = {
  OR = {
    realm_allows_female_courtiers_trigger = yes
    has_law = status_of_women_4
    AND = {
      religion_group = pagan_group
      NOT = { has_religion_feature = religion_patriarchal }
    }
    religion = cathar
    religion = messalian
    religion_allows_female_temple_holders = yes
  }
}

# Whether any male role would be CLICKABLE on ER.11 -- at least one role's
# per-option trigger (its sex-gate AND its affordability) passes. The decision
# allow and the effect's gender-routing chain use this so the player can never
# be routed into an empty role picker. The lax _eligible_trigger above only
# checks sex-gates; this tightens to "can we actually afford something".
elite_recruitment_any_male_role_eligible_and_affordable_trigger = {
  OR = {
    AND = {
      # Marshal/Steward/Chancellor/Spymaster share the standard male sex-gate.
      NOT = { has_religion_feature = religion_matriarchal }
      OR = {
        elite_recruitment_marshal_can_afford_trigger = yes
        elite_recruitment_steward_can_afford_trigger = yes
        elite_recruitment_chancellor_can_afford_trigger = yes
        elite_recruitment_spymaster_can_afford_trigger = yes
      }
    }
    AND = {
      # Male chaplain has the widened gate (matriarchal faiths may still ordain men).
      OR = {
        religion_allows_male_temple_holders = yes
        has_religion_feature = religion_patriarchal
      }
      elite_recruitment_chaplain_can_afford_trigger = yes
    }
  }
}

# Female mirror. The three branches reflect the per-option gates inside ER.12:
# (a) standard female gate for marshal/steward/chancellor (gender law);
# (b) widened spymaster gate (pagan-non-patriarchal / Cathar / Messalian);
# (c) female chaplain faith-ordination gate. Each branch is AND'd with the
# matching role's can-afford trigger, so the combined OR is true only when
# at least one specific role is BOTH eligible AND affordable.
elite_recruitment_any_female_role_eligible_and_affordable_trigger = {
  OR = {
    AND = {
      OR = {
        realm_allows_female_courtiers_trigger = yes
        has_law = status_of_women_4
      }
      OR = {
        elite_recruitment_marshal_can_afford_trigger = yes
        elite_recruitment_steward_can_afford_trigger = yes
        elite_recruitment_chancellor_can_afford_trigger = yes
      }
    }
    AND = {
      OR = {
        realm_allows_female_courtiers_trigger = yes
        has_law = status_of_women_4
        AND = {
          religion_group = pagan_group
          NOT = { has_religion_feature = religion_patriarchal }
        }
        religion = cathar
        religion = messalian
      }
      elite_recruitment_spymaster_can_afford_trigger = yes
    }
    AND = {
      OR = {
        religion_allows_female_temple_holders = yes
        has_religion_feature = religion_matriarchal
      }
      elite_recruitment_chaplain_can_afford_trigger = yes
    }
  }
}

# Whether women may hold court roles regardless of the status_of_women laws
realm_allows_female_courtiers_trigger = {
  OR = {
    has_game_rule = { name = gender value = all }
    has_religion_feature = religion_matriarchal
    has_religion_feature = religion_equal
    has_religion_feature = religion_feature_bon
  }
}

# Whether the character belongs to any Holy Fury warrior lodge
in_warrior_lodge_trigger = {
  OR = {
    society_member_of = warrior_lodge_norse
    society_member_of = warrior_lodge_tengri
    society_member_of = warrior_lodge_baltic
    society_member_of = warrior_lodge_finnish
    society_member_of = warrior_lodge_slavic
    society_member_of = warrior_lodge_west_african
    society_member_of = warrior_lodge_zun
    society_member_of = warrior_lodge_bon
    society_member_of = warrior_lodge_hellenic
  }
}

# --- Per-role affordability gates ---
# Each role's check mirrors its pay_elite_*_cost effect. Gold needs both
# scaled_wealth (income fraction) and wealth (the firm floor); the secondary currency
# matches the role. Modicum is the default and the rule-absent fallback.

# Marshal: balanced gold and Prestige.
elite_recruitment_marshal_can_afford_trigger = {
  trigger_if = {
    limit = { has_game_rule = { name = elite_recruitment value = pittance } }
    scaled_wealth = 0.125
    wealth = 8
    prestige = 25
  }
  trigger_else_if = {
    limit = { has_game_rule = { name = elite_recruitment value = trifle } }
    scaled_wealth = 0.25
    wealth = 15
    prestige = 50
  }
  trigger_else_if = {
    limit = { has_game_rule = { name = elite_recruitment value = bounty } }
    scaled_wealth = 1.0
    wealth = 60
    prestige = 200
  }
  trigger_else_if = {
    limit = { has_game_rule = { name = elite_recruitment value = fortune } }
    scaled_wealth = 2.0
    wealth = 120
    prestige = 400
  }
  trigger_else = { # modicum
    scaled_wealth = 0.5
    wealth = 30
    prestige = 100
  }
}

# --- Menu pagination helpers ---
# True when all 5 male recruit options would be visible simultaneously (every
# sex-gate passes and every role's affordability check passes). The role picker
# uses this to switch between single-page (<=4 visible) and paginated (all 5
# visible) layouts: when true, Spymaster and Chaplain hide from page 1 and the
# "Or perhaps another..." button shows; when false, page 1 shows whichever
# roles are visible directly with no nav button.
elite_recruitment_all_5_male_visible_trigger = {
  NOT = { has_religion_feature = religion_matriarchal }
  OR = {
    religion_allows_male_temple_holders = yes
    has_religion_feature = religion_patriarchal
  }
  elite_recruitment_marshal_can_afford_trigger = yes
  elite_recruitment_steward_can_afford_trigger = yes
  elite_recruitment_chancellor_can_afford_trigger = yes
  elite_recruitment_spymaster_can_afford_trigger = yes
  elite_recruitment_chaplain_can_afford_trigger = yes
}

# Female mirror.

  }

  }

  elite_recruitment_chancellor_can_afford_trigger = yes
  elite_recruitment_spymaster_can_afford_trigger = yes
  elite_recruitment_chaplain_can_afford_trigger = yes
}
