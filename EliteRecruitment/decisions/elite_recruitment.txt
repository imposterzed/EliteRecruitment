##################################################
##              Elite Recruitment               ##
##################################################
## The Recruit an Elite Specialist decision.    ##
## Opens the ER.10/11/12 menu chain so the      ##
## player picks the recruit's sex and council   ##
## role; auto-skips the gender picker when      ##
## only one sex has any eligible role.          ##
##################################################

decisions = {

  recruit_elite_specialist = {
    ai = no

    potential = {
      elite_recruitment_enabled_trigger = yes
      OR = {
        elite_recruitment_any_male_role_eligible_trigger = yes
        elite_recruitment_any_female_role_eligible_trigger = yes
      }
    }

    allow = {
      in_seclusion = no
      OR = {
        elite_recruitment_marshal_can_afford_trigger = yes
        elite_recruitment_steward_can_afford_trigger = yes
        elite_recruitment_chancellor_can_afford_trigger = yes
        elite_recruitment_spymaster_can_afford_trigger = yes
        elite_recruitment_chaplain_can_afford_trigger = yes
      }
    }

    effect = {
      custom_tooltip = {
        text = recruit_elite_specialist_TT

        # Auto-skip the gender picker when only one sex has eligible roles.
        # Otherwise open ER.10 to ask the player.
        if = {
          limit = {
            elite_recruitment_any_male_role_eligible_trigger = yes
            elite_recruitment_any_female_role_eligible_trigger = yes
          }
          character_event = { id = ER.10 }
        }
        else_if = {
          limit = { elite_recruitment_any_male_role_eligible_trigger = yes }
          character_event = { id = ER.11 }
        }
        else = {
          character_event = { id = ER.12 }
        }
      }
    }

    ai_will_do = {
      factor = 0
    }
  }

}
