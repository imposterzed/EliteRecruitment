namespace = ER

############################################################
##            Elite Recruitment Arrival Event             ##
############################################################
##                                                        ##
## ER.1 = recruit-arrival pop-up, offering to seat the    ##
## new recruit on their council.                          ##
##                                                        ##
############################################################

# Fired on the recruiting ruler by every recruit decision, after the recruit is
# created and saved as event_target:invited_character. The recruit carries an
# elite_recruit_<role> flag, so only that role's options show. If a seat is
# already filled the current holder is dismissed first; the dismissal portrait
# and penalty only render when someone actually holds the job. The bonus minor
# titles only appear when a slot is free.

    name = {
      text = ER_1_marshal
      trigger = {
        NOR = {
          any_courtier = { has_job_title = job_marshal }
          any_vassal = { has_job_title = job_marshal }
        }
      }
    }
    trigger = {
      event_target:invited_character = {
        has_character_flag = elite_recruit_marshal
        OR = {
          elite_recruitment_gender_gating_bypassed_trigger = yes
          can_be_marshal_trigger = yes
        }
      }
    }
    # If the seat is filled, dismiss the holder first; this block's portrait and
    # penalty only render when someone actually holds the job, so an empty seat stays clean.
    if = {
      limit = { any_courtier = { has_job_title = job_marshal } }
      custom_tooltip = { text = ER_1_replace }
      any_courtier = {
        limit = { has_job_title = job_marshal }
        opinion = { modifier = opinion_fired_from_council who = ROOT years = 5 }
        remove_title = job_marshal
      }
    }
    if = {
      limit = { any_vassal = { has_job_title = job_marshal } }
      custom_tooltip = { text = ER_1_replace }
      any_vassal = {
        limit = { has_job_title = job_marshal }
        opinion = { modifier = opinion_fired_from_council who = ROOT years = 5 }
        remove_title = job_marshal
      }
    }
    event_target:invited_character = {
      show_scope_change = no
      give_job_title = job_marshal
    }

  # --- Marshal bonus: a field command, only while a command slot is free (vanilla caps) ---
  option = {
    name = ER_1_command
    trigger = {
      event_target:invited_character = { has_character_flag = elite_recruit_marshal }
      trigger_if = {
        limit = { real_tier = COUNT }
        NOT = { any_realm_character = { count = 2 liege = { character = ROOT } has_minor_title = title_commander } }
      }
      trigger_else_if = {
        limit = { real_tier = DUKE }
        NOT = { any_realm_character = { count = 4 liege = { character = ROOT } has_minor_title = title_commander } }
      }
      trigger_else_if = {
        limit = { real_tier = KING  is_nomadic = no }
        NOT = { any_realm_character = { count = 6 liege = { character = ROOT } has_minor_title = title_commander } }
      }
      trigger_else_if = {
        limit = { real_tier = EMPEROR  is_nomadic = no }
        NOT = { any_realm_character = { count = 8 liege = { character = ROOT } has_minor_title = title_commander } }
      }
      trigger_else_if = {
        limit = { real_tier = KING  is_nomadic = yes }
        NOT = { any_realm_character = { count = 2 liege = { character = ROOT } has_minor_title = title_commander } }
      }
      trigger_else_if = {
        limit = { real_tier = EMPEROR  is_nomadic = yes }
        NOT = { any_realm_character = { count = 4 liege = { character = ROOT } has_minor_title = title_commander } }
      }
      trigger_else = { always = no }
    }
    event_target:invited_character = {
      show_scope_change = no
      give_minor_title = title_commander
    }
  }

  # --- Chaplain bonus: court physician, if a trained physician and the post is open (Reaper's Due) ---
  option = {
    name = ER_1_physician
    trigger = {
      has_dlc = "Reapers"
      event_target:invited_character = {
        has_character_flag = elite_recruit_chaplain
        trait = physician
      }
      NOT = { any_realm_character = { liege = { character = ROOT } has_minor_title = title_court_physician } }
    }
    custom_tooltip = { text = ER_1_physician_suited }
    event_target:invited_character = {
      show_scope_change = no
      give_minor_title = title_court_physician
