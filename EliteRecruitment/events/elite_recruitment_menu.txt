# Menu events: Recruit an Elite Specialist
# ER.10 = gender picker; ER.11 / ER.12 = per-sex job pickers. The decision
# opens ER.10 when both sexes have eligible roles, else it auto-skips
# directly to ER.11 (male) or ER.12 (female).

# ER.10 -- Gender picker. Only fired when both sexes have at least one
# eligible role; the decision auto-skips this event otherwise. No per-option
# trigger needed: the auto-skip guarantees both options have a populated
# menu to lead to when ER.10 fires.
character_event = {
  id = ER.10

  is_triggered_only = yes # recruit_elite_specialist decision
  desc = EVTDESC_ER_10
  picture = GFX_event_elite_recruitment_menu_gender

  option = {
    name = ER_10_male
    character_event = { id = ER.11 }
  }

  option = {
    name = ER_10_female
    character_event = { id = ER.12 }
  }

  option = {
    name = ER_10_cancel
  }
}

# ER.11 -- Male role picker. One event handles both single-page (<=4 roles
# visible) and paginated (all 5 visible) layouts: Spymaster and Chaplain hide
# here when all 5 would be visible at once (deferred to ER.13), and the
# "Or perhaps another..." button appears in their place. Otherwise the visible
# roles show directly with no nav button. Roles list in council order
# (Chancellor, Marshal, Steward, Spymaster, Chaplain). Marshal/Steward/Chancellor
# /Spymaster share the "NOT matriarchal" sex-gate; the chaplain uses the
# faith-ordination gate instead, independent of liege patriarchy.
character_event = {
  id = ER.11

  is_triggered_only = yes # recruit_elite_specialist decision / ER.10 male option / ER.13 back button
  desc = EVTDESC_ER_11
  picture = GFX_event_elite_recruitment_menu_role_male

  # Chancellor
  option = {
    name = ER_11_chancellor
    trigger = {
      OR = {
        elite_recruitment_gender_gating_bypassed_trigger = yes
        elite_recruitment_generic_male_gate_trigger = yes
      }
      elite_recruitment_chancellor_can_afford_trigger = yes
    }

    pay_elite_chancellor_cost_effect = yes

    custom_tooltip = {
      text = recruit_chancellor_TT

      create_male_chancellor_dynasty_lottery_effect = yes

      new_character = {
        assign_random_adult_age_effect = yes
        assign_chancellor_base_attributes_effect = yes
        assign_chancellor_traits_effect = yes
      }
    }

    character_event = { id = ER.1 }
  }

  # Marshal
  option = {
    name = ER_11_marshal
    trigger = {
      OR = {
        elite_recruitment_gender_gating_bypassed_trigger = yes
        elite_recruitment_generic_male_gate_trigger = yes
      }
      elite_recruitment_marshal_can_afford_trigger = yes
    }

    pay_elite_marshal_cost_effect = yes

    custom_tooltip = {
      text = recruit_marshal_TT

      create_male_marshal_dynasty_lottery_effect = yes

      new_character = {
        assign_random_adult_age_effect = yes
        assign_marshal_base_attributes_effect = yes
        assign_marshal_traits_effect = yes
      }
    }

    character_event = { id = ER.1 }
  }

  # Steward
  option = {
    name = ER_11_steward
    trigger = {
      OR = {
        elite_recruitment_gender_gating_bypassed_trigger = yes
        elite_recruitment_generic_male_gate_trigger = yes
      }
      elite_recruitment_steward_can_afford_trigger = yes
    }

    pay_elite_steward_cost_effect = yes

    custom_tooltip = {
      text = recruit_steward_TT

      create_male_steward_dynasty_lottery_effect = yes

      new_character = {
        assign_random_adult_age_effect = yes
        assign_steward_base_attributes_effect = yes
        assign_steward_traits_effect = yes
      }
    }

    character_event = { id = ER.1 }
  }

  # Spymaster -- hidden when all 5 are visible (deferred to ER.13)
  option = {
    name = ER_11_spymaster
    trigger = {
      OR = {
        elite_recruitment_gender_gating_bypassed_trigger = yes
        elite_recruitment_generic_male_gate_trigger = yes
      }
      elite_recruitment_spymaster_can_afford_trigger = yes
      NOT = { elite_recruitment_all_5_male_visible_trigger = yes }
    }

    pay_elite_spymaster_cost_effect = yes

    custom_tooltip = {
      text = recruit_spymaster_TT

      create_male_spymaster_dynasty_lottery_effect = yes

      new_character = {
        assign_random_adult_age_effect = yes
        assign_spymaster_base_attributes_effect = yes
        assign_spymaster_traits_effect = yes
      }
    }

    character_event = { id = ER.1 }
  }

  # Chaplain -- faith-ordination, not liege patriarchy. Hidden when all 5
  # are visible (deferred to ER.13).
  option = {
    name = ER_11_chaplain
    trigger = {
      OR = {
        elite_recruitment_gender_gating_bypassed_trigger = yes
        elite_recruitment_chaplain_male_gate_trigger = yes
      }
      elite_recruitment_chaplain_can_afford_trigger = yes
      NOT = { elite_recruitment_all_5_male_visible_trigger = yes }
    }

    pay_elite_chaplain_cost_effect = yes

    custom_tooltip = {
      text = recruit_chaplain_TT

      create_male_chaplain_dynasty_lottery_effect = yes

      new_character = {
        assign_random_adult_age_effect = yes
        assign_chaplain_base_attributes_effect = yes
        assign_chaplain_traits_effect = yes
      }
    }

    character_event = { id = ER.1 }
  }

  # Or perhaps another... -- shown only when all 5 are visible, forcing
  # pagination to ER.13 where Spymaster and Chaplain wait.
  option = {
    name = ER_11_more
    trigger = { elite_recruitment_all_5_male_visible_trigger = yes }
    character_event = { id = ER.13 }
  }
}
