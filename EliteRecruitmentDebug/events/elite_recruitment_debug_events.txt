##################################################
##       Elite Recruitment Debug Events         ##
##################################################
## Dev/test events. Not part of the shipped     ##
## feature surface â€” never wired to a decision  ##
## or on_action. Fire from the in-game console: ##
##   event ERD.<id> <charID>                    ##
##                                              ##
## Useful when an effect we depend on can only  ##
## be reached via a long vanilla chain (e.g.    ##
## warrior lodge legendary journey for the      ##
## legendary_commander_bloodline). Each event   ##
## here is a one-shot, no event_target deps,    ##
## hide_window so it fires silently.            ##
##                                              ##
## Add a new debug grant by following the same  ##
## pattern: character_event { id = ERD.N        ##
## hide_window = yes is_triggered_only = yes    ##
## immediate = { <effect> } }                   ##
##################################################

namespace = ERD


# ERD.1 â€” Grant legendary_commander_bloodline (carries the
# bloodline_inspire_commanders flag). Use to test the marshal
# legendary build path in assign_marshal_legendary_chance,
# which gates on has_bloodline_flag = bloodline_inspire_commanders
# and rolls 5% per marshal recruit for the legendary archetype.
#
# Console:  event ERD.1 <charID>
character_event = {
  id = ERD.1
  hide_window = yes
  is_triggered_only = yes

  immediate = {
    if = {
      limit = { is_female = yes }
      create_bloodline = {
        type = legendary_commander_bloodline
        inheritance = matrilineal
      }
    }
    else = {
      create_bloodline = { type = legendary_commander_bloodline }
    }
  }
}
