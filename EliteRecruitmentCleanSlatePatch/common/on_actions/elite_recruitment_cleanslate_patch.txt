##################################################
##     Elite Recruitment - CleanSlate Patch      ##
##################################################
## Sets a global flag at game start so the base ##
## mod uses CleanSlate's renamed trait names.    ##
##################################################

# on_startup is additive, so it won't override other mods' startup actions.
on_startup = {
  effect = {
    set_global_flag = elite_recruitment_cleanslate
  }
}
